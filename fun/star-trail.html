星轨涂鸦

Canvas 星轨粒子涂鸦：鼠标 / 触屏拖拽留下渐变星轨，颜色随时间循环流动。

拖拽涂鸦 · 颜色循环

按住并拖动鼠标（或手指）绘画；星轨粒子会自动减速消散，形成拖尾效果。

使用体验

  1. 在画布上按住拖动，会持续喷出渐变色粒子并留下光轨，松手后粒子减速消散。
  2. “模式”可在星轨 / 辉光两种效果间切换；辉光模式叠加多重圆形渐变，更柔和。
  3. 支持触屏（touch-action: none 防页面滚动），移动端同样可玩。

代码要点

js
const parts = [];
let drawing = false, hue = 0;

function paint(x, y) {
  // 每次移动发射 3 个粒子，带随机速度 / 寿命
  for (let i = 0; i < 3; i++) {
    parts.push({
      x, y,
      vx: (Math.random() - 0.5) * 1.6,
      vy: (Math.random() - 0.5) * 1.6,
      life: 1,
      hue: (hue + Math.random() * 40) % 360
    });
  }
}

function step() {
  ctx.clearRect(0, 0, W, H);          // 全清画法（粒子自绘拖尾）
  for (let i = parts.length - 1; i >= 0; i--) {
    const p = parts[i];
    p.x += p.vx; p.y += p.vy;
    p.vx *= 0.98; p.vy *= 0.98;        // 减速 = 拖尾
    p.life -= 0.018;
    ctx.fillStyle = `hsla(${p.hue}, 80%, 60%, ${Math.max(0, p.life)})`;
    ctx.beginPath();
    ctx.arc(p.x, p.y, 3 * p.life + 0.6, 0, Math.PI * 2);
    ctx.fill();
    if (p.life <= 0) parts.splice(i, 1); // 回收死亡粒子
  }
  requestAnimationFrame(step);
}
