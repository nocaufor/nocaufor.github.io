密码生成器与强度检测

生成强随机密码、实时评估强度与熵值，附弱密码清单提示，本地运行零上传。

场景：安全工具 技术栈：HTML / JS 来源：GitHub 参考

项目简介

密码是账号安全的第一道防线。本页提供两段可独立使用的代码：强随机密码生成（可指定长度与字符集）与强度评估（根据长度、字符种类与熵值给出等级与改进建议）。全部在浏览器本地计算，密码不会离开设备。

代码示例：强随机密码生成

js
function randomInt(n) {
  const a = new Uint32Array(1);
  crypto.getRandomValues(a);
  return a[0] % n;
}

const SETS = {
  lower: "abcdefghijklmnopqrstuvwxyz",
  upper: "ABCDEFGHIJKLMNOPQRSTUVWXYZ",
  digit: "0123456789",
  symbol: "!@#$%^&*()-_=+[]{};:,.<>?"
};

function generatePassword(length, use = ["lower", "upper", "digit", "symbol"]) {
  const pool = use.map(k => SETS[k]).join("");
  if (!pool) throw new Error("empty charset");
  const chars = [];
  // 保证每种已选字符集至少出现一次，再随机补齐
  for (const k of use) chars.push(SETS[k][randomInt(SETS[k].length)]);
  while (chars.length < length) chars.push(pool[randomInt(pool.length)]);
  // Fisher-Yates 洗牌，避免首字符规律
  for (let i = chars.length - 1; i > 0; i--) {
    const j = randomInt(i + 1);
    [chars[i], chars[j]] = [chars[j], chars[i]];
  }
  return chars.join("");
}

代码示例：强度评估与熵值

js
function entropyBits(pw) {
  let poolSize = 0;
  if (/[a-z]/.test(pw)) poolSize += 26;
  if (/[A-Z]/.test(pw)) poolSize += 26;
  if (/[0-9]/.test(pw)) poolSize += 10;
  if (/[^a-zA-Z0-9]/.test(pw)) poolSize += 32;
  return poolSize ? pw.length * Math.log2(poolSize) : 0;
}

function strengthLevel(pw) {
  const bits = entropyBits(pw);
  if (pw.length < 8 || bits < 45) return "weak";
  if (bits < 70) return "medium";
  if (bits < 100) return "strong";
  return "excellent";
}

// 常用弱密码黑名单（示例，可扩充）
const WEAK = new Set(["123456", "password", "qwerty", "admin", "111111"]);
function isWeak(pw) {
  const low = pw.toLowerCase();
  return WEAK.has(low) || /(.)\1{3,}/.test(pw) || low.includes("123");
}

操作步骤

  1. 按需选择长度与字符集，点击“生成”得到随机密码，密码生成使用 crypto.getRandomValues，勿用 Math.random。
  2. 粘贴任意密码到强度检测框，实时查看熵值与强度等级；若命中弱密码黑名单会明确提示。
  3. 生产环境建议配合密码管理器存储，并启用两步验证，单靠强密码不足以防钓鱼。
