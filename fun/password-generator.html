项目简介

密码是账号安全的第一道防线。本页提供两段可独立使用的代码：强随机密码生成（可指定长度与字符集）与强度评估（根据长度、字符种类与熵值给出等级与改进建议）。全部在浏览器本地计算，密码不会离开设备。

代码示例：强随机密码生成

js
function randomInt(n) {
  const a = new Uint32Array(1);
  crypto.getRandomValues(a);
  return a[0] % n;
}

const SETS = {
  lower: "abcdefghijklmnopqrstuvwxyz",
  upper: "ABCDEFGHIJKLMNOPQRSTUVWXYZ",
  digit: "0123456789",
  symbol: "!@#$%^&*()-_=+[]{};:,.<>?"
};

function generatePassword(length, use = ["lower", "upper", "digit", "symbol"]) {
  const pool = use.map(k => SETS[k]).join("");
  if (!pool) throw new Error("empty charset");
  const chars = [];
  // 保证每种已选字符集至少出现一次，再随机补齐
  for (const k of use) chars.push(SETS[k][randomInt(SETS[k].length)]);
  while (chars.length < length) chars.push(pool[randomInt(pool.length)]);
  // Fisher-Yates 洗牌，避免首字符规律
  for (let i = chars.length - 1; i > 0; i--) {
    const j = randomInt(i + 1);
    [chars[i], chars[j]] = [chars[j], chars[i]];
  }
  return chars.join("");
}

代码示例：强度评估与熵值

js
function entropyBits(pw) {
  let poolSize = 0;
  if (/[a-z]/.test(pw)) poolSize += 26;
  if (/[A-Z]/.test(pw)) poolSize += 26;
  if (/[0-9]/.test(pw)) poolSize += 10;
  if (/[^a-zA-Z0-9]/.test(pw)) poolSize += 32;
  return poolSize ? pw.length * Math.log2(poolSize) : 0;
}

function strengthLevel(pw) {
  const bits = entropyBits(pw);
  if (pw.length < 8 || bits < 45) return "weak";
  if (bits < 70) return "medium";
  if (bits < 100) return "strong";
  return "excellent";
}

// 常用弱密码黑名单（示例，可扩充）
const WEAK = new Set(["123456", "password", "qwerty", "admin", "111111"]);
function isWeak(pw) {
  const low = pw.toLowerCase();
  return WEAK.has(low) || /(.)\1{3,}/.test(pw) || low.includes("123");
}

操作步骤

  1. 在演示区设置长度与字符集，点击"生成密码"，下方结果区一次输出 1/5/10 条候选；"复制第一条"可一键写入剪贴板。
  2. 生成过程中每类已勾选字符至少出现一次（拒绝采样保证无模偏差），"排除易混字符"可去掉 0 O 1 l I | 这些容易看错的字符。
  3. 把任意密码粘贴进"强度检测"框：实时给出信息熵（bit）、强度等级、离线爆破估算与具体改进建议；命中常见弱密码黑名单会直接标红提示。
  4. 对比测试：输入 abc123（命中黑名单）与 MyDog$2024!atHome（约 100 bit）观察等级与条形图变化。

app.js

安全随机（拒绝采样 + Fisher-Yates 洗牌）、熵值评估、黑名单与离线爆破估算；含 Node 导出分支。

js
/* 密码生成与强度评估：无依赖，直接运行
   随机源使用 Web Crypto（crypto.getRandomValues），带拒绝采样消除模偏差 */

const SIMILAR = "0O1lI|`'\"";
const SETS = {
  upper: "ABCDEFGHIJKLMNOPQRSTUVWXYZ",
  lower: "abcdefghijklmnopqrstuvwxyz",
  digit: "0123456789",
  symbol: "!@#$%^&*()-_=+[]{};:,.?/~"
};

function buildPool(opts) {
  let pool = "";
  if (opts.upper) pool += SETS.upper;
  if (opts.lower) pool += SETS.lower;
  if (opts.digit) pool += SETS.digit;
  if (opts.symbol) pool += SETS.symbol;
  if (opts.noSimilar) pool = [...pool].filter((c) => !SIMILAR.includes(c)).join("");
  return pool;
}

/* 无模偏差的随机下标：拒绝落在尾部不成整倍数的区间 */
function randomIndex(max) {
  if (max <= 0) throw new Error("empty pool");
  const limit = Math.floor(0x100000000 / max) * max;
  const buf = new Uint32Array(1);
  let v;
  do { crypto.getRandomValues(buf); v = buf[0]; } while (v >= limit);
  return v % max;
}

/* 保证每类已勾选字符至少出现一次，再整体洗牌 */
function generate(length, opts) {
  const pool = buildPool(opts);
  if (!pool) throw new Error("请至少选择一类字符");
  const required = [];
  ["upper", "lower", "digit", "symbol"].forEach((k) => {
    if (!opts[k]) return;
    const chars = [...SETS[k]].filter((c) => !(opts.noSimilar && SIMILAR.includes(c)));
    if (chars.length) required.push(chars[randomIndex(chars.length)]);
  });
  const out = required.slice(0, length);
  while (out.length < length) out.push(pool[randomIndex(pool.length)]);
  for (let i = out.length - 1; i > 0; i--) {           /* Fisher-Yates 洗牌 */
    const j = randomIndex(i + 1);
    [out[i], out[j]] = [out[j], out[i]];
  }
  return out.join("");
}

const WEAK = ["password", "123456", "12345678", "qwerty", "111111", "abc123", "admin",
  "letmein", "iloveyou", "welcome", "monkey", "dragon", "666666", "888888"];

/* 熵值（bit）：按实际字符分布估计字符池大小 */
function entropy(pw) {
  if (!pw) return 0;
  let pool = 0;
  if (/[a-z]/.test(pw)) pool += 26;
  if (/[A-Z]/.test(pw)) pool += 26;
  if (/[0-9]/.test(pw)) pool += 10;
  if (/[^A-Za-z0-9]/.test(pw)) pool += 30;
  const unique = new Set(pw).size;
  if (unique <= 2) pool = Math.min(pool, 4);           /* aaa111 这类模式 |
                                                          可猜空间远小于字符池 */
  return Math.round(pw.length * Math.log2(Math.max(pool, 2)) * 10) / 10;
}

function evaluate(pw) {
  const bits = entropy(pw);
  const advices = [];
  const lower = pw.toLowerCase();
  const hitWeak = WEAK.some((w) => lower === w || (pw.length >= 6 && lower.includes(w)));

  if (pw.length < 12) advices.push("长度不足 12 位：长度是抵抗暴力破解最有效的因素，建议 16 位以上。");
  if (!/[A-Z]/.test(pw)) advices.push("缺少大写字母，字符池变小。");
  if (!/[0-9]/.test(pw)) advices.push("缺少数字。");
  if (!/[^A-Za-z0-9]/.test(pw)) advices.push("缺少符号，建议至少加入一个符号。");
  if (/(.)\1{2,}/.test(pw)) advices.push("存在连续重复字符（如 aaa），会被字典规则优先命中。");
  if (/^(?:\d+|[a-z]+|[A-Z]+)$/.test(pw)) advices.push("只由单一类型字符组成，模式过于规律。");
  if (/(19|20)\d{2}/.test(pw)) advices.push("包含年份，属于社工字典常见组合。");
  if (hitWeak) advices.push("命中常见弱密码/黑名单片段，必须更换。");

  let level = "很弱", pct = 8;
  if (hitWeak) { level = "已被泄露的弱密码"; pct = 5; }
  else if (bits >= 90) { level = "极强"; pct = 100; }
  else if (bits >= 70) { level = "强"; pct = 78; }
  else if (bits >= 50) { level = "中等"; pct = 52; }
  else if (bits >= 35) { level = "偏弱"; pct = 28; }
  else { level = "很弱"; pct = 10; }
  if (!advices.length) advices.push("未发现明显弱点。建议配合密码管理器保存，并为重要账号开启两步验证。");

  /* 离线爆破估算：1e11 次/秒（高端 GPU 集群量级） */
  const trials = Math.pow(2, bits) / 2;
  const seconds = trials / 1e11;
  return { bits, level, pct, advices, crack: humanTime(seconds) };
}

function humanTime(sec) {
  if (sec < 1) return "瞬间";
  const units = [["年", 31557600], ["天", 86400], ["小时", 3600], ["秒", 1]];
  const out = [];
  let rest = sec;
  for (const [name, size] of units) {
    if (rest >= size && out.length < 2) { out.push(Math.floor(rest / size) + " " + name); rest %= size; }
  }
  return out.join(" ") || "1 秒内";
}

/* ------------------------------ 界面绑定 ------------------------------ */
function opts() {
  return {
    upper: document.getElementById("upper").checked,
    lower: document.getElementById("lower").checked,
    digit: document.getElementById("digit").checked,
    symbol: document.getElementById("symbol").checked,
    noSimilar: document.getElementById("noSimilar").checked
  };
}

function renderEval() {
  const pw = document.getElementById("test").value;
  const r = evaluate(pw);
  document.getElementById("bar").style.width = r.pct + "%";
  document.getElementById("summary").textContent =
    pw ? `熵值 ${r.bits} bit · ${r.level} · 离线爆破约 ${r.crack}` : "请输入密码";
  const ul = document.getElementById("advice");
  ul.innerHTML = "";
  r.advices.forEach((a) => { const li = document.createElement("li"); li.textContent = a; ul.appendChild(li); });
}

if (typeof document !== "undefined") {
  document.addEventListener("DOMContentLoaded", () => {
  const len = document.getElementById("len");
  len.addEventListener("input", () => { document.getElementById("lenVal").textContent = len.value; });
  document.getElementById("gen").addEventListener("click", () => {
    document.getElementById("out").textContent = generate(Number(len.value), opts());
  });
  document.getElementById("test").addEventListener("input", renderEval);
  document.getElementById("gen").click();
  renderEval();
  });
}

/* ---- 供 Node 最小自测使用（浏览器中该分支不执行） ---- */
if (typeof module !== "undefined" && module.exports) {
  module.exports = {
    SETS: SETS, SIMILAR: SIMILAR, buildPool: buildPool, randomIndex: randomIndex,
    generate: generate, entropy: entropy, evaluate: evaluate
  };
}

# 密码生成器与强度检测（password-generator）

零依赖单页工具：安全随机密码生成 + 实时强度评估。所有计算在浏览器本地完成。

## 运行

1. 双击 index.html 即可使用（无需构建、无需依赖）。
2. 生成使用 crypto.getRandomValues + 拒绝采样，避免 Math.random 的可预测性与取模偏差。

## 安全边界（务必了解）

- 阈值参考：熵值 >= 70 bit 视为"强"，>= 90 bit 为"极强"；低于 50 bit 不建议用于重要账号。
- 离线爆破估算按 1e11 次/秒（高端 GPU 集群量级）折算，仅用于比较密码之间强弱，不代表真实攻击时间。
- 生成页面本身不应通过 HTTP 明文访问（避免中间人窃取），建议本地打开或 HTTPS 部署。
- 强度评估不能发现所有风险（如撞库泄露、钓鱼），重要账号请配合密码管理器 + 两步验证。

## 依赖清单与最小自测

- 依赖清单：零第三方依赖（浏览器原生 Web Crypto + DOM）；Node 18+ 可直接跑逻辑自测，不需要浏览器。
- 最小自测（复制本目录后执行，期望输出 `20 true 已被泄露的弱密码`）：

```bash
node -e "const m=require('./app.js');const p=m.generate(20,{upper:true,lower:true,digit:true,symbol:true,noSimilar:true});console.log(p.length, m.entropy(p)>100, m.evaluate('123456').level)"
```
运行标注：可直接运行（浏览器 Web Crypto + DOM，零依赖、无需构建）。需先改的配置项：无必改项；如需调整默认长度/字符集，改 index.html 中对应控件默认值即可。最小自测：index.html 打开即用；逻辑自测用 README.md 里的 node 命令（期望 20 true 已被泄露的弱密码）。
