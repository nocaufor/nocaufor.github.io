代码示例：评分组件

js
// components/rating/rating.js —— 可复用评分组件
Component({
  properties: {
    value: { type: Number, value: 0 },   // 当前分数
    max:   { type: Number, value: 5 },   // 满分
    size:  { type: Number, value: 24 }   // 星号尺寸 rpx
  },
  data: { list: [1, 2, 3, 4, 5] },
  methods: {
    onTap(e) {
      const n = e.currentTarget.dataset.n;
      this.setData({ value: n });
      this.triggerEvent("change", { value: n });
    }
  }
});

// components/debounce-btn/debounce-btn.js —— 防抖按钮
Component({
  properties: { text: { type: String, value: "提交" } },
  data: { locked: false },
  methods: {
    onTap() {
      if (this.data.locked) return;      // 已锁定则忽略
      this.setData({ locked: true });    // 300ms 内防重入
      this.triggerEvent("tap");
      setTimeout(() => this.setData({ locked: false }), 300);
    }
  }
});

操作步骤

  1. 在 components/rating/ 目录创建 rating.json 声明组件，rating.wxml/js/wxss 实现。
  2. 页面 usingComponents 注册后即可 <rating value="4"> 使用。
  3. 评分变化通过 triggerEvent("change") 通知父页面。
  4. 防抖按钮组件避免支付/提交被重复点击。

components/rating/rating.js

评分组件：属性、事件与逻辑层的桥接。

javascript
// components/rating/rating.js —— 可复用评分组件
// 纯逻辑放在同目录 rating-core.js，便于在电脑上用 node 直接单测（tools/selftest.js）
const core = require('./rating-core.js');

Component({
  properties: {
    value: { type: Number, value: 0, observer(v) { this.setData({ list: core.starList(this.data.max), percent: core.percent(v, this.data.max) }); } },
    max: { type: Number, value: 5, observer(v) { this.setData({ list: core.starList(v), percent: core.percent(this.data.value, v) }); } },
    size: { type: Number, value: 24 },          // 星号尺寸（rpx 基准 ×2）
    readonly: { type: Boolean, value: false }
  },

  data: {
    list: [1, 2, 3, 4, 5],
    percent: 0
  },

  lifetimes: {
    attached() {
      this.setData({ list: core.starList(this.data.max), percent: core.percent(this.data.value, this.data.max) });
    }
  },

  methods: {
    onTap(e) {
      if (this.data.readonly) return;
      const n = Number(e.currentTarget.dataset.n);
      const next = core.toggle(this.data.value, n, this.data.max);
      this.setData({ value: next, percent: core.percent(next, this.data.max) });
      this.triggerEvent('change', { value: next, percent: this.data.percent });
    }
  }
});

components/rating/rating-core.js

评分的纯逻辑层（UMD）：toggle / percent / starList，含边界与除零保护。

javascript
/**
 * rating-core.js —— 评分组件的纯逻辑层（UMD）
 * 之所以拆出纯逻辑：小程序端 require 使用，电脑端可用 node 直接 require 做单元测试，
 * 无需启动微信开发者工具即可验证“最小用例”。
 */
(function (root, factory) {
  var api = factory();
  if (typeof module === 'object' && module.exports) {
    module.exports = api;            // Node.js / 小程序 CommonJS
  } else {
    root.RatingCore = api;           // 浏览器/其他环境兜底
  }
})(typeof globalThis !== 'undefined' ? globalThis : this, function () {
  'use strict';

  function clamp(v, lo, hi) {
    v = Number(v);
    if (isNaN(v)) return lo;
    return v < lo ? lo : (v > hi ? hi : v);
  }

  /** 点击第 index 颗星（1-based）：点当前值则清零，否则设为该值 */
  function toggle(current, index, max) {
    var cur = clamp(current, 0, max);
    var idx = clamp(index, 0, max);
    return idx === cur ? 0 : idx;
  }

  /** 评分百分比，用于宽度类样式（如进度条） */
  function percent(value, max) {
    if (!max || max <= 0) return 0;
    return Math.round(clamp(value, 0, max) / max * 100);
  }

  /** 生成星星列表（视图层 wx:for 使用） */
  function starList(max) {
    var list = [];
    for (var i = 1; i <= max; i++) list.push(i);
    return list;
  }

  return { clamp: clamp, toggle: toggle, percent: percent, starList: starList };
});

components/debounce-button/debounce-button.js

防抖按钮：leading + trailing 合并、可选 throttle。

javascript
// components/debounce-button/debounce-button.js —— 防抖按钮组件
// 逻辑复用 debounce-core.js（可被 node 单测），组件只负责 UI 与事件派发。
const { createDebouncer } = require('./debounce-core.js');

Component({
  properties: {
    text: { type: String, value: '提交' },
    wait: { type: Number, value: 800 },        // 防抖窗口（ms）
    size: { type: String, value: 'normal' },   // normal | small
    disabled: { type: Boolean, value: false }
  },

  data: {
    pressed: 0,      // 实际触发次数（被合并的点击不计数）
    tapped: 0,       // 用户点击次数（含被合并的）
    hint: ''
  },

  lifetimes: {
    attached() {
      this._deb = createDebouncer(this.properties.wait, () => this._emit(), {});
      this._count = 0;
    },
    detached() {
      if (this._deb) this._deb.cancel();
    }
  },

  methods: {
    _emit() {
      this._count += 1;
      this.setData({ pressed: this._count, hint: '已触发 ' + this._count + ' 次' });
      this.triggerEvent('action', { count: this._count, wait: this.properties.wait });
    },

    onTap() {
      if (this.properties.disabled) return;
      const fired = this._deb.call();
      this.setData({
        tapped: this.data.tapped + 1,
        hint: fired ? '立即触发' : '窗口内合并，等待尾随触发'
      });
    }
  }
});

components/debounce-button/debounce-core.js

防抖纯逻辑层（UMD）：可注入定时器，便于 Node 下逐帧自测。

javascript
/**
 * debounce-core.js —— 防抖/节流纯逻辑层（UMD）
 * 采用「leading + trailing」策略：窗口内首次点击立即生效，窗口内的后续点击被合并为
 * 最后一次尾随执行，避免用户狂点导致重复提交。
 *
 * 时间与定时器均支持注入（opts.now / opts.setTimeout / opts.clearTimeout），
 * 因此可以在 node 中用假时钟做确定性单元测试，无需真实等待。
 */
(function (root, factory) {
  var api = factory();
  if (typeof module === 'object' && module.exports) {
    module.exports = api;
  } else {
    root.DebounceCore = api;
  }
})(typeof globalThis !== 'undefined' ? globalThis : this, function () {
  'use strict';

  function createDebouncer(wait, fn, opts) {
    opts = opts || {};
    var now = opts.now || function () { return Date.now(); };
    var setT = opts.setTimeout || function (cb, ms) { return setTimeout(cb, ms); };
    var clearT = opts.clearTimeout || function (id) { clearTimeout(id); };

    var timer = null;
    var last = -Infinity;      // 上次真正执行的时间
    var pendingArgs = null;

    function invoke(args) {
      last = now();
      timer = null;
      pendingArgs = null;
      fn.apply(null, args || []);
    }

    return {
      /** 返回 true 表示本次调用立即执行，false 表示被合并等待尾随执行 */
      call: function () {
        var args = Array.prototype.slice.call(arguments);
        var t = now();
        if (t - last >= wait) {
          invoke(args);                   // leading：窗口外的点击直接生效
          return true;
        }
        pendingArgs = args;               // 窗口内：记录最后一次参数
        if (timer) clearT(timer);
        timer = setT(function () { if (pendingArgs) invoke(pendingArgs); }, wait - (t - last));
        return false;
      },
      /** 取消未决的尾随执行（组件销毁时调用） */
      cancel: function () {
        if (timer) clearT(timer);
        timer = null;
        pendingArgs = null;
      },
      get pending() { return !!timer; }
    };
  }

  /** 节流：固定周期首调用立即执行，周期内其余调用直接丢弃（不做尾随） */
  function throttle(wait, fn, now) {
    now = now || function () { return Date.now(); };
    var last = -Infinity;
    return function () {
      var t = now();
      if (t - last < wait) return false;
      last = t;
      fn.apply(null, arguments);
      return true;
    };
  }

  return { createDebouncer: createDebouncer, throttle: throttle };
});

tools/selftest.js

Node 自测脚本：15 项断言覆盖评分边界、防抖窗口与 cancel/throttle。

javascript
#!/usr/bin/env node
/**
 * tools/selftest.js —— 电脑端最小用例验证（无需微信开发者工具）
 * 运行：node tools/selftest.js   （或 npm test）
 *
 * 覆盖：评分逻辑（toggle/percent/starList/clamp）与防抖逻辑（leading+trailing、
 * 窗口内合并、cancel、throttle），全部通过输出 ALL CHECKS PASSED。
 */
'use strict';

const rating = require('../components/rating/rating-core.js');
const { createDebouncer, throttle } = require('../components/debounce-button/debounce-core.js');

let failed = 0;
function check(name, cond, extra) {
  console.log('  [' + (cond ? 'OK' : 'FAIL') + '] ' + name + (cond ? '' : '  => ' + JSON.stringify(extra)));
  if (!cond) failed++;
}

/** 可注入的假时钟，让防抖测试不依赖真实等待 */
function makeClock() {
  let t = 0, seq = 0;
  const timers = new Map();
  return {
    now: () => t,
    setTimeout: (fn, ms) => { const id = ++seq; timers.set(id, { fn, at: t + ms }); return id; },
    clearTimeout: (id) => { timers.delete(id); },
    tick: (ms) => {
      t += ms;
      Array.from(timers.entries()).forEach(([id, o]) => {
        if (o.at <= t) { timers.delete(id); o.fn(); }
      });
    },
    get size() { return timers.size; }
  };
}

console.log('rating-core：');
check('toggle(3,3,5) 点击当前值应清零', rating.toggle(3, 3, 5) === 0, rating.toggle(3, 3, 5));
check('toggle(0,4,5) 应设为 4', rating.toggle(0, 4, 5) === 4, rating.toggle(0, 4, 5));
check('toggle(3,9,5) 越界应被 clamp 到 5', rating.toggle(3, 9, 5) === 5, rating.toggle(3, 9, 5));
check('percent(3,5) = 60', rating.percent(3, 5) === 60, rating.percent(3, 5));
check('percent(5,0) 除零保护 = 0', rating.percent(5, 0) === 0, rating.percent(5, 0));
check('starList(5) 长度 5', rating.starList(5).length === 5, rating.starList(5));
check('clamp(NaN,0,5) 回退下界 0', rating.clamp(NaN, 0, 5) === 0, rating.clamp(NaN, 0, 5));

console.log('debounce-core（leading + trailing）：');
{
  const c = makeClock();
  let n = 0;
  const deb = createDebouncer(100, () => n++, c);
  const r1 = deb.call();
  const r2 = deb.call();
  const r3 = deb.call();
  check('窗口外首击立即执行', r1 === true && n === 1, { r1, n });
  check('窗口内连击被合并（未立即执行）', r2 === false && r3 === false && n === 1, { r2, r3, n });
  c.tick(100);
  check('尾随执行合并为 1 次', n === 2, { n });
  c.tick(50);
  check('尾随执行后无残留定时器', c.size === 0, c.size);
  c.tick(100);
  const r4 = deb.call();
  check('超过窗口后再次立即执行', r4 === true && n === 3, { r4, n });
}

console.log('debounce-core（cancel / throttle）：');
{
  const c = makeClock();
  let n = 0;
  const deb = createDebouncer(100, () => n++, c);
  deb.call(); deb.call();
  deb.cancel();
  c.tick(500);
  check('cancel 后尾随不再执行', n === 1 && c.size === 0, { n, size: c.size });
}
{
  const c = makeClock();
  let n = 0;
  const th = throttle(100, () => n++, c.now);
  const a = th(), b = th();
  c.tick(120);
  const d = th();
  check('throttle 周期内丢弃、周期后放行', a === true && b === false && d === true && n === 2, { a, b, d, n });
}

console.log(failed === 0 ? 'ALL CHECKS PASSED' : 'SELF TEST FAILED (' + failed + ')');
process.exit(failed === 0 ? 0 : 1);
