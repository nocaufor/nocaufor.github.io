/* app.js —— 虚拟陪伴：规则引擎 + 亲密度（纯逻辑部分可在 node 中自测） */
var RULES = {
  warm: {
    greet: ["你来啦，今天过得怎么样？", "我等你好一会儿了。"],
    care: ["别太累了，记得喝水。", "你要照顾好自己，我才放心。"],
    echo: ["嗯，我在听。", "然后呢？我想继续听。"],
    like: ["和你聊天这件事本身就很值得。", "你说的话我都记着呢。"]
  },
  sharp: {
    greet: ["哟，还知道回来？", "忙完了？我以为你把我删了。"],
    care: ["少熬夜，别指望我夸你。", "喝口水吧，别硬撑。"],
    echo: ["就这样？", "行吧，勉强算个话题。"],
    like: ["勉强承认，你有点意思。", "别得意，我可不是随便夸人。"]
  },
  calm: {
    greet: ["你好，今天还顺利吗？", "在的，慢慢说。"],
    care: ["注意休息，节奏比速度重要。", "别把弦绷太紧。"],
    echo: ["我记下了。", "这件事可以再展开说说。"],
    like: ["和你说话很舒服。", "这种交流很有意义。"]
  }
};
var STYLES = {
  daily: { tail: "～" },
  literary: { tail: "。" }
};

/* classify：极简意图识别（关键词打分） */
function classify(text) {
  var t = String(text);
  if (/^(你好|hi|hello|在吗|嗨)/i.test(t.trim())) return "greet";
  if (/(累|难过|烦|压力|失眠|加班|生病|难受)/.test(t)) return "care";
  if (/(喜欢|爱你|想你|喜欢我|陪我)/.test(t)) return "like";
  return "echo";
}

/* reply：规则引擎主入口（纯函数，node 可测） */
function reply(text, opts) {
  opts = opts || {};
  var persona = opts.persona || "warm";
  var style = opts.style || "daily";
  var intimacy = opts.intimacy == null ? 0 : opts.intimacy;
  var kind = classify(text);
  var bank = RULES[persona][kind];
  var idx = Math.abs(hash(text + persona)) % bank.length;
  var line = bank[idx];
  if (style === "literary") line = line.replace(/([^。！？])$/, "$1。");
  else line = line.replace(/([^。！？～])$/, "$1" + STYLES.daily.tail);
  if (intimacy >= 8 && kind === "echo") line += "（我们聊了不少了，我很高兴。）";
  return { kind: kind, text: line, intimacy: intimacy + (kind === "care" || kind === "like" ? 2 : 1) };
}

function hash(s) {
  var h = 2166136261;
  for (var i = 0; i < s.length; i++) { h ^= s.charCodeAt(i); h = Math.imul(h, 16777619); }
  return h;
}

/* 大模型适配层：填入 API Key 即可切换到真实模型（默认不启用） */
async function llmReply(text, cfg) {
  var res = await fetch(cfg.endpoint, {
    method: "POST",
    headers: { "Content-Type": "application/json", "Authorization": "Bearer " + cfg.apiKey },
    body: JSON.stringify({
      model: cfg.model,
      messages: [
        { role: "system", content: "你的性格：" + cfg.persona + "；语言风格：" + cfg.style },
        { role: "user", content: text }
      ]
    })
  });
  if (!res.ok) throw new Error("接口返回 " + res.status);
  var data = await res.json();
  return data.choices[0].message.content;
}

if (typeof module !== "undefined" && module.exports) {
  module.exports = { RULES: RULES, classify: classify, reply: reply, llmReply: llmReply };
}

if (typeof document !== "undefined") {
  document.addEventListener("DOMContentLoaded", function () {
    var chat = document.getElementById("chat"), input = document.getElementById("input"),
        sendBtn = document.getElementById("send"), personaSel = document.getElementById("persona"),
        styleSel = document.getElementById("style"), engineBtn = document.getElementById("engine"),
        barsEl = document.getElementById("bars"), hint = document.getElementById("hint");
    var intimacy = 0, useLLM = false, cfg = { endpoint: "https://api.openai.com/v1/chat/completions", apiKey: "", model: "gpt-4o-mini", persona: "温柔", style: "日常口语" };

    var row = document.createElement("div");
    row.className = "bar";
    row.innerHTML = "<span>亲密度</span><span><i id='bar-int'></i></span><span id='val-int'>0</span>";
    barsEl.appendChild(row);

    function bubble(text, who) {
      var d = document.createElement("div");
      d.className = who;
      d.textContent = text;
      chat.appendChild(d);
      chat.scrollTop = chat.scrollHeight;
    }
    function paint() {
      document.getElementById("bar-int").style.width = Math.min(100, intimacy * 6) + "%";
      document.getElementById("val-int").textContent = intimacy;
    }
    async function send() {
      var text = input.value.trim();
      if (!text) return;
      bubble(text, "me");
      input.value = "";
      if (useLLM && cfg.apiKey) {
        try {
          var out = await llmReply(text, cfg);
          bubble(out, "pet");
        } catch (e) {
          bubble("（大模型接口调用失败：" + e.message + "，已回落本地规则）", "pet");
          bubble(reply(text, { persona: personaSel.value, style: styleSel.value, intimacy: intimacy }).text, "pet");
        }
      } else {
        var r = reply(text, { persona: personaSel.value, style: styleSel.value, intimacy: intimacy });
        intimacy = r.intimacy;
        bubble(r.text, "pet");
      }
      paint();
    }
    sendBtn.addEventListener("click", send);
    input.addEventListener("keydown", function (e) { if (e.key === "Enter") send(); });
    engineBtn.addEventListener("click", function () {
      useLLM = !useLLM;
      engineBtn.textContent = useLLM ? "引擎：大模型 API" : "引擎：本地规则";
      hint.textContent = useLLM
        ? "已切换到大模型引擎：需在 app.js 的 cfg 中填入 endpoint / apiKey / model，否则自动回落本地规则。"
        : "当前为本地规则引擎：无网络、无 API Key 也能跑。";
    });
    personaSel.addEventListener("change", function () { bubble("（性格已切换为：" + personaSel.options[personaSel.selectedIndex].text + "）", "pet"); });
    styleSel.addEventListener("change", function () { bubble("（风格已切换为：" + styleSel.options[styleSel.selectedIndex].text + "）", "pet"); });
    bubble(RULES.warm.greet[0], "pet");
    paint();
  });
}

README.md

运行方式与接大模型说明

markdown
# 虚拟对象 · 人味儿对话陪伴（独立可运行工程）

作品站 fun 区「虚拟对象」的独立工程版：内置规则引擎开箱即用，可选接入大模型 API。

## 文件
- `index.html`：性格 / 风格 / 引擎切换 + 对话面板
- `app.js`：意图识别（关键词打分）→ 性格语料库 → 风格化 → 亲密度累计；附大模型适配层
- `styles.css`：样式

## 运行
1. 双击 `index.html`，直接开始对话（无需网络、无需 Key）
2. 逻辑自测：`node -e "const m=require('./app.js');console.log(m.classify('今天好累'), m.reply('今天好累',{persona:'warm'}).kind)"` → 期望 `care care`

## 接大模型（可选）
在 `app.js` 的 `cfg` 中填入 `endpoint` / `apiKey` / `model`，再点「引擎」按钮切换；接口失败会自动回落本地规则。
- 兼容 OpenAI Chat Completions 协议的服务均可（如各家云厂商的兼容网关）
- 注意：浏览器直连会暴露 Key，生产环境应改为由你的后端转发

## 扩展点
- 换语料：改 `RULES[性格][意图]` 数组即可，意图分类器在 `classify()`
- 加长期记忆：把对话写入 localStorage，回复前检索相关历史片段拼进 prompt

运行方式

  1. 新建一个目录，把上面全部文件按文件名分别保存（用「一键复制整份工程」可一次拿全）。
  2. 前端部分：直接双击 index.html 即可运行；若浏览器限制本地文件读取，执行 python -m http.server 8000 后访问 http://localhost:8000/。
  3. 逻辑自测：各文件 README.md 中给出了对应的 Node / Python 自测命令（均为真实可跑的最小用例）。

标注：可直接运行、无需构建；要改造只需动 app.js 顶部常量与 styles.css 里的变量。

复制结果为 # ===== 文件名 ===== 分隔的多文件文本，按文件名分别保存即可；也可逐块用右上角“复制代码”单独取用。
