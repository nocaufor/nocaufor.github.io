习惯打卡与连续天数

每日打卡、连续坚持天数统计与日历热力图，localStorage 本地存储。

场景：效率工具 技术栈：HTML / JS 来源：GitHub 参考

项目简介

把目标拆成每天可重复的小动作，是养成习惯最朴素的方法。本页提供打卡数据模型与连续天数算法：用 localStorage 持久化日期集合，支持补卡判断、断签重置和日历热力图展示，可直接嵌入个人效率工具。

const KEY = "nocau-habit-log";
const today = () => new Date().toISOString().slice(0, 10); // YYYY-MM-DD

function loadLog() {
  try { return JSON.parse(localStorage.getItem(KEY)) || []; } catch (e) { return []; }
}
function saveLog(log) { localStorage.setItem(KEY, JSON.stringify(log)); }

function toggleCheck(date = today()) {
  const log = loadLog();
  const i = log.indexOf(date);
  if (i >= 0) log.splice(i, 1); else log.push(date);
  saveLog(log);
  return i < 0;
}

代码示例：连续天数与热力图

js
const DAY = 86400000;
function parse(d) { return new Date(d + "T00:00:00Z").getTime(); }

// 从最近一天往前数连续打卡天数（昨天中断即清零）
function streakDays(log) {
  const set = new Set(log);
  let streak = 0;
  let cursor = parse(today());
  while (set.has(new Date(cursor).toISOString().slice(0, 10))) {
    streak += 1;
    cursor -= DAY;
  }
  return streak;
}

// 近 28 周热力图数据：week 数组内 7 天是否打卡
function heatmap(log, weeks = 28) {
  const set = new Set(log);
  const rows = [];
  const end = parse(today()) + DAY;
  for (let w = 0; w < weeks; w++) {
    const week = [];
    for (let d = 0; d < 7; d++) {
      const t = end - (weeks - w) * 7 * DAY + d * DAY;
      const key = new Date(t).toISOString().slice(0, 10);
      week.push({ date: key, done: set.has(key), future: t >= end });
    }
    rows.push(week);
  }
  return rows;
}

app.js

数据模型、连续天数算法、最长连续、完成率与热力图渲染，全部为纯函数 + 一个 render 入口。

js
/* 习惯打卡：数据模型 + 连续天数算法 + 热力图渲染（无依赖，可直接运行） */
const KEY = "nocau-habit-log";
const DAY = 86400000;

const todayStr = () => new Date().toISOString().slice(0, 10);
const parse = (d) => new Date(d + "T00:00:00Z").getTime();
const fmt = (t) => new Date(t).toISOString().slice(0, 10);

function loadLog() {
  try { return JSON.parse(localStorage.getItem(KEY)) || []; } catch (e) { return []; }
}
function saveLog(log) {
  log.sort();
  localStorage.setItem(KEY, JSON.stringify(log));
}

/* 打卡 / 撤销：返回 true 表示本次为新增打卡 */
function toggleCheck(date = todayStr()) {
  const log = loadLog();
  const i = log.indexOf(date);
  if (i >= 0) log.splice(i, 1); else log.push(date);
  saveLog(log);
  return i < 0;
}

/* 当前连续：从今天往前数，允许今天未打卡时从昨天起算 */
function currentStreak(log, ref = todayStr()) {
  const set = new Set(log);
  let cursor = parse(ref);
  if (!set.has(fmt(cursor))) cursor -= DAY;      // 今天未打卡不打断连击
  let streak = 0;
  while (set.has(fmt(cursor))) { streak += 1; cursor -= DAY; }
  return streak;
}

/* 历史最长连续 */
function bestStreak(log) {
  const days = [...new Set(log)].map(parse).sort((a, b) => a - b);
  let best = 0, run = 0, prev = null;
  for (const t of days) {
    run = prev !== null && t - prev === DAY ? run + 1 : 1;
    prev = t;
    if (run > best) best = run;
  }
  return best;
}

/* 近 weeks 周热力图：列=周（周一起），行=周一..周日 */
function heatmap(log, weeks = 28) {
  const set = new Set(log);
  const today = parse(todayStr());
  const dow = (new Date(today).getUTCDay() + 6) % 7;        // 周一=0
  const end = today + (6 - dow) * DAY;                      // 本周周日
  const start = end - (weeks * 7 - 1) * DAY;
  const rows = [];
  for (let w = 0; w < weeks; w++) {
    const col = [];
    for (let d = 0; d < 7; d++) {
      const t = start + (w * 7 + d) * DAY;
      col.push({ date: fmt(t), done: set.has(fmt(t)), future: t > today, today: t === today });
    }
    rows.push(col);
  }
  return rows;
}

/* 近 28 周完成率（不含未来日期） */
function completionRate(log, weeks = 28) {
  const set = new Set(log);
  const grid = heatmap(log, weeks).flat().filter((c) => !c.future);
  if (!grid.length) return 0;
  return Math.round((grid.filter((c) => c.done).length / grid.length) * 100);
}

function render() {
  const log = loadLog();
  const name = document.getElementById("name").value.trim() || "未命名习惯";
  document.getElementById("streak").textContent = currentStreak(log);
  document.getElementById("best").textContent = bestStreak(log);
  document.getElementById("total").textContent = log.length;
  document.getElementById("rate").textContent = completionRate(log) + "%";

  const heat = document.getElementById("heat");
  heat.innerHTML = "";
  heatmap(log).forEach((col) =>
    col.forEach((c) => {
      const cell = document.createElement("div");
      cell.className = "cell" + (c.done ? " on" : "") + (c.today ? " today" : "");
      cell.title = c.date + (c.future ? "（未来）" : c.done ? " 已打卡" : " 未打卡");
      heat.appendChild(cell);
    })
  );
  document.getElementById("status").textContent =
    `「${name}」累计 ${log.length} 天，当前连续 ${currentStreak(log)} 天。`;
}

if (typeof document !== "undefined") {
  document.addEventListener("DOMContentLoaded", () => {
  const dateInput = document.getElementById("date");
  dateInput.value = todayStr();
  document.getElementById("check").addEventListener("click", () => {
    const on = toggleCheck(dateInput.value || todayStr());
    document.getElementById("status").textContent = on ? "已记录打卡。" : "已撤销该日打卡。";
    render();
  });
  document.getElementById("name").addEventListener("input", render);
  render();
});
}


/* Node 端自测导出：浏览器中 module 未定义，此段不会执行 */
if (typeof module !== "undefined" && module.exports) {
  module.exports = { KEY, todayStr, toggleCheck, loadLog, saveLog,
                     currentStreak, bestStreak, heatmap, completionRate };
}

# 习惯打卡与连续天数（habit-tracker）

零依赖的浏览器端习惯追踪小工具：localStorage 持久化、连续天数统计、GitHub 风格热力图。

## 运行

1. 直接双击 index.html（或拖入浏览器）即可使用，无需构建、无需依赖。
2. 数据保存在浏览器 localStorage 的 `nocau-habit-log` 键，清空浏览器数据会一并清除。

## 自测（零依赖，Node 14+）

```bash
# 连续天数 / 历史最长连续 两个纯算法的最小用例
node -e "const m=require('./app.js');console.log(m.currentStreak(['2026-09-18','2026-09-19'],'2026-09-19'),m.bestStreak(['2026-09-01','2026-09-02','2026-09-03']))"
# 期望输出：2 3
```

浏览器端无需任何命令：双击 `index.html` 即是完整功能。

## 二次开发

- 多习惯：把 KEY 改为 `nocau-habit-log:<习惯名>`，或存对象 `{ habit: [dates] }`。
- 导出数据：`JSON.stringify(loadLog())` 可一键导出；恢复时 `saveLog(arr)` 写回。
- 对接后端：把 loadLog / saveLog 换成 fetch 调用即可，其余算法无需改动。
